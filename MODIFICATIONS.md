### 1. histories 테이블 추가

```sql
CREATE TABLE IF NOT EXISTS histories (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT TIMEZONE('utc', NOW()),
    teacher_id TEXT NOT NULL,
    client_id TEXT NOT NULL,
    client_name TEXT NOT NULL,
    type TEXT NOT NULL,              -- 'session', 'consultation', 'noshow', 'reschedule', 'delete'
    category TEXT NOT NULL,          -- 'session_history', 'consultation_history'
    action TEXT NOT NULL,            -- '수업시작', '수업종료', '상담시작', '상담종료', '일정변경', '미출석', '삭제'
    date TEXT NOT NULL,
    start_time TEXT,
    end_time TEXT,
    duration_minutes INT,
    original_date TEXT,
    changed_date TEXT,
    original_time TEXT,
    changed_time TEXT,
    details TEXT,
    memo TEXT
);
```

## 🔧 관리자 페이지 (admin.html) 수정사항

### NoticesModule 컴포넌트 수정

#### 1. 공지사항 수정 기능
```javascript
const [editingId, setEditingId] = useState(null);

const handleEdit = (notice) => {
    setEditingId(notice.id);
    setForm({ type: notice.type, title: notice.title, content: notice.content });
    setIsWriting(true);
};

const handleSave = async () => {
    if (editingId) {
        // 수정 모드
        const payload = { ...form, updated_at: new Date().toISOString() };
        const { error } = await supabase.from('notices').update(payload).eq('id', editingId);
        if (!error) {
            setNotices(notices.map(n => n.id === editingId ? {...n, ...payload} : n));
            setEditingId(null);
        }
    } else {
        // 신규 등록 (기존 로직)
    }
    setIsWriting(false);
    setForm({ type: '일반', title: '', content: '' });
};
```

#### 2. 공지사항 삭제 기능
```javascript
const handleDelete = async (id) => {
    if(confirm('정말로 이 공지사항을 삭제하시겠습니까?')) {
        const { error } = await supabase.from('notices').delete().eq('id', id);
        if (!error) {
            setNotices(notices.filter(n => n.id !== id));
        }
    }
};
```

#### 3. 공지사항 숨기기/보이기 기능
```javascript
const handleToggleHidden = async (id, currentHidden) => {
    const { error } = await supabase.from('notices')
        .update({ is_hidden: !currentHidden })
        .eq('id', id);
    if (!error) {
        setNotices(notices.map(n => n.id === id ? {...n, isHidden: !currentHidden} : n));
    }
};
```

#### 4. UI 버튼 추가
```javascript
// 리스트 항목에 편집/삭제/숨기기 버튼 추가
<div className="flex gap-2">
    <button onClick={() => handleEdit(notice)}>
        <Edit3 size={16}/>
    </button>
    <button onClick={() => handleDelete(notice.id)}>
        <Trash2 size={16}/>
    </button>
    <button onClick={() => handleToggleHidden(notice.id, notice.isHidden)}>
        {notice.isHidden ? <Eye size={16}/> : <EyeOff size={16}/>}
    </button>
</div>
```

### 2. 프로필 삭제 시 미완료 스케줄 일괄 삭제

```javascript
const handleDeleteClient = async (clientId) => {
    if (confirm('이 프로필을 삭제하시겠습니까? 미완료된 모든 수업/상담도 함께 삭제됩니다.')) {
        // 1. 상태를 'deleted'로 변경
        await supabase.from('clients').update({ status: 'deleted' }).eq('id', clientId);
        
        // 2. 미완료 스케줄 삭제 (트리거에서 자동으로 histories에 기록됨)
        await supabase.from('schedules')
            .delete()
            .eq('client_id', clientId)
            .in('status', ['scheduled', 'in-progress']);
        
        // 3. 삭제 이력 생성
        await supabase.from('histories').insert({
            teacher_id: user.teacherId,
            client_id: clientId,
            client_name: client.name,
            type: 'delete',
            category: client.type === 'reserve' ? 'consultation_history' : 'session_history',
            action: '프로필삭제',
            date: new Date().toISOString().split('T')[0],
            details: '프로필이 삭제되어 모든 미완료 일정이 삭제됨'
        });
        
        onDelete(clientId);
    }
};
```

### 4. 상담예약자 → 내담자 전환 시 이력 이전

```javascript
const handleConvertClient = async (client) => {
    if (confirm(`${client.name}님을 정규 내담자로 전환하시겠습니까?`)) {
        // 1. 타입 변경
        await supabase.from('clients').update({ type: 'client' }).eq('id', client.id);
        
        // 2. 상담 이력을 수업 이력으로 변경
        await supabase.from('histories')
            .update({ category: 'session_history' })
            .eq('client_id', client.id)
            .eq('category', 'consultation_history');
        
        // 3. 전환 이력 생성
        await supabase.from('histories').insert({
            teacher_id: user.teacherId,
            client_id: client.id,
            client_name: client.name,
            type: 'conversion',
            category: 'session_history',
            action: '내담자전환',
            date: new Date().toISOString().split('T')[0],
            details: '상담예약자에서 정규 내담자로 전환됨. 기존 상담이력이 수업이력으로 이동됨.'
        });
        
        showToast(`${client.name}님이 정규 내담자로 등록되었습니다.`);
    }
};
```
